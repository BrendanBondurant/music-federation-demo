artists :4001

extend schema @link(url: "https://specs.apollo.dev/federation/v2.3", import: ["@key"])

type Artist @key(fields: "id") {
  id: ID!
  name: String!
  kind: ArtistKind!
  instruments: [String!]!
  styles: [String!]!
  bio: String
  members: [Membership!]!
  memberOf: [Membership!]!
}

enum ArtistKind {
  PERSON
  ENSEMBLE
}

type Membership {
  group: Artist!
  member: Artist!
  role: String
}

type Query {
  artist(
    id: ID!
  ): Artist
  artists(
    instrument: String
  ): [Artist!]!
}

catalog :4002

extend schema @link(url: "https://specs.apollo.dev/federation/v2.3", import: ["@key"])

interface Piece @key(fields: "id") {
  id: ID!
  title: String!
  genre: Genre!
  musicalKey: String
}

enum Genre {
  CLASSICAL
  JAZZ
  FLAMENCO
  BLUEGRASS
}

type Work @key(fields: "id") {
  id: ID!
  title: String!
  catalogNumber: String
  composer: Artist
  genre: Genre!
  movements: [Movement!]!
}

type Movement implements Piece @key(fields: "id") {
  id: ID!
  title: String!
  position: Int
  work: Work!
  genre: Genre!
  musicalKey: String
}

type Tune implements Piece @key(fields: "id") {
  id: ID!
  title: String!
  style: String
  composer: Artist
  contrafactOf: Tune
  contrafacts: [Tune!]!
  genre: Genre!
  musicalKey: String
}

type Artist @key(fields: "id") {
  id: ID!
  composedPieces: [Piece!]!
  composedWorks: [Work!]!
}

type Query {
  piece(
    id: ID!
  ): Piece
  pieces(
    musicalKey: String
  ): [Piece!]!
  tune(
    id: ID!
  ): Tune
  tunes: [Tune!]!
  work(
    id: ID!
  ): Work
  works(
    composer: ID
  ): [Work!]!
}

discography :4003

extend schema @link(url: "https://specs.apollo.dev/federation/v2.3", import: ["@key", "@interfaceObject"])

type Album @key(fields: "id") {
  id: ID!
  title: String!
  year: Int
  label: String
  leaders: [Artist!]!
  credits: [Credit!]!
  tracks: [Recording!]!
}

type Credit {
  artist: Artist!
  role: String
}

type Recording @key(fields: "id") {
  id: ID!
  piece: Piece!
  album: Album
  performers: [Artist!]!
  performanceKey: String
  source: String
  notes: String
  bpm: Int
}

type Piece @key(fields: "id") @interfaceObject {
  id: ID!
  recordings: [Recording!]!
}

type Artist @key(fields: "id") {
  id: ID!
  albums: [Album!]!
  recordings(
    piece: ID
  ): [Recording!]!
}

type Query {
  album(
    id: ID!
  ): Album
  albums: [Album!]!
}
